    <script>
        function startBirthday() {
            const nameInput = document.getElementById("username").value.trim();
            
            if (nameInput === "") {
                alert("Please enter a name first!");
                return;
            }

            // Hide input form and show greeting
            document.getElementById("input-section").classList.add("hidden");
            document.getElementById("greeting-section").classList.remove("hidden");

            // Update heading with the user's name
            document.getElementById("birthday-message").innerText = `Happy Birthday, ${nameInput}! 🎉`;

            // Play the song
            const song = document.getElementById("birthday-song");
            song.play().catch(error => {
                console.log("Audio autoplay restricted by browser: ", error);
            });
        }
    </script>
